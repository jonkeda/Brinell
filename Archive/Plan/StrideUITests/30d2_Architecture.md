## 2.3 Communication Architecture

### 2.3.1 Channel Abstraction

```csharp
/// <summary>
/// Abstraction for communication between test process and game process.
/// </summary>
public interface IAutomationChannel : IDisposable
{
    /// <summary>
    /// Connect to the game's automation service.
    /// </summary>
    Task ConnectAsync(TimeSpan timeout);
    
    /// <summary>
    /// Check if connected to game.
    /// </summary>
    bool IsConnected { get; }
    
    /// <summary>
    /// Send a command and receive response.
    /// </summary>
    Task<AutomationResponse> SendCommandAsync(AutomationCommand command);
    
    /// <summary>
    /// Disconnect from game.
    /// </summary>
    Task DisconnectAsync();
}
```

### 2.3.2 Named Pipe Implementation

```csharp
public class NamedPipeChannel : IAutomationChannel
{
    private const string PipeName = "Oravey.Automation";
    private NamedPipeClientStream? _pipeClient;
    private readonly JsonSerializerOptions _jsonOptions;
    
    public async Task ConnectAsync(TimeSpan timeout)
    {
        _pipeClient = new NamedPipeClientStream(".", PipeName, 
            PipeDirection.InOut, PipeOptions.Asynchronous);
        
        await _pipeClient.ConnectAsync((int)timeout.TotalMilliseconds);
    }
    
    public async Task<AutomationResponse> SendCommandAsync(AutomationCommand command)
    {
        if (_pipeClient == null || !_pipeClient.IsConnected)
            throw new InvalidOperationException("Not connected to game");
        
        // Serialize and send command
        var json = JsonSerializer.Serialize(command, _jsonOptions);
        var buffer = Encoding.UTF8.GetBytes(json + "\n");
        await _pipeClient.WriteAsync(buffer);
        await _pipeClient.FlushAsync();
        
        // Read response
        var responseBuffer = new byte[8192];
        var bytesRead = await _pipeClient.ReadAsync(responseBuffer);
        var responseJson = Encoding.UTF8.GetString(responseBuffer, 0, bytesRead);
        
        return JsonSerializer.Deserialize<AutomationResponse>(responseJson, _jsonOptions)!;
    }
}
```

### 2.3.3 In-Process Channel (Alternative)

For simpler testing without separate processes:

```csharp
public class InProcessChannel : IAutomationChannel
{
    private readonly AutomationService _automationService;
    
    public InProcessChannel(AutomationService automationService)
    {
        _automationService = automationService;
    }
    
    public Task ConnectAsync(TimeSpan timeout)
    {
        // Already connected - same process
        return Task.CompletedTask;
    }
    
    public async Task<AutomationResponse> SendCommandAsync(AutomationCommand command)
    {
        // Execute on game thread
        return await _automationService.ExecuteCommandAsync(command);
    }
}
```

## 2.4 Game-Side Automation Architecture

### 2.4.1 AutomationService

Central registry for testable UI elements:

```csharp
/// <summary>
/// Central service for UI test automation in the game.
/// Conditionally compiled for test builds only.
/// </summary>
#if ENABLE_AUTOMATION
public class AutomationService : ScriptComponent
{
    private readonly ConcurrentDictionary<string, WeakReference<UIElement>> _elements = new();
    private readonly ConcurrentDictionary<string, WeakReference<IPageObject>> _pages = new();
    
    /// <summary>
    /// Register a UI element for automation access.
    /// </summary>
    public void RegisterElement(string automationId, UIElement element)
    {
        _elements[automationId] = new WeakReference<UIElement>(element);
        Log.Debug("Registered automation element: {AutomationId}", automationId);
    }
    
    /// <summary>
    /// Get a registered UI element.
    /// </summary>
    public UIElement? GetElement(string automationId)
    {
        if (_elements.TryGetValue(automationId, out var weakRef) 
            && weakRef.TryGetTarget(out var element))
        {
            return element;
        }
        return null;
    }
    
    /// <summary>
    /// Query element state.
    /// </summary>
    public ElementState GetElementState(string automationId)
    {
        var element = GetElement(automationId);
        if (element == null)
            return new ElementState { Exists = false };
        
        return new ElementState
        {
            Exists = true,
            IsVisible = element.IsVisible,
            IsEnabled = element.IsEnabled,
            Bounds = GetElementBounds(element),
            Text = GetElementText(element)
        };
    }
}
#endif
```

### 2.4.2 AutomationHost

Named pipe server running in the game:

```csharp
#if ENABLE_AUTOMATION
public class AutomationHost : ScriptComponent
{
    private const string PipeName = "Oravey.Automation";
    private NamedPipeServerStream? _pipeServer;
    private CancellationTokenSource? _cancellationSource;
    private readonly AutomationService _automationService;
    
    public override async Task Execute()
    {
        _cancellationSource = new CancellationTokenSource();
        
        while (!_cancellationSource.Token.IsCancellationRequested)
        {
            try
            {
                _pipeServer = new NamedPipeServerStream(PipeName, 
                    PipeDirection.InOut, 1, PipeTransmissionMode.Byte, 
                    PipeOptions.Asynchronous);
                
                await _pipeServer.WaitForConnectionAsync(_cancellationSource.Token);
                
                await HandleClientAsync(_pipeServer, _cancellationSource.Token);
            }
            catch (OperationCanceledException)
            {
                break;
            }
            finally
            {
                _pipeServer?.Dispose();
            }
        }
    }
    
    private async Task HandleClientAsync(NamedPipeServerStream pipe, CancellationToken ct)
    {
        using var reader = new StreamReader(pipe);
        using var writer = new StreamWriter(pipe) { AutoFlush = true };
        
        while (!ct.IsCancellationRequested && pipe.IsConnected)
        {
            var commandJson = await reader.ReadLineAsync();
            if (string.IsNullOrEmpty(commandJson)) continue;
            
            var command = JsonSerializer.Deserialize<AutomationCommand>(commandJson);
            
            // Execute command on game thread
            AutomationResponse response = null!;
            await Script.Scheduler.Add(() =>
            {
                response = ProcessCommand(command!);
                return Task.CompletedTask;
            });
            
            await writer.WriteLineAsync(JsonSerializer.Serialize(response));
        }
    }
}
#endif
```

### 2.4.3 UI Element Registration

Automatic registration during UI creation:

```csharp
// Extension method for easy registration
public static class AutomationExtensions
{
    public static T WithAutomationId<T>(this T element, string automationId) 
        where T : UIElement
    {
#if ENABLE_AUTOMATION
        // Store automation ID
        element.SetAutomationId(automationId);
        
        // Register with service (lazy initialization)
        AutomationService.Instance?.RegisterElement(automationId, element);
#endif
        return element;
    }
}

// Usage in UI creation
var button = new Button
{
    Content = new TextBlock { Text = "New Game" },
    BackgroundColor = new Color(60, 60, 80, 200)
}.WithAutomationId("MainMenu.NewGame");
```

## 2.5 Test Execution Flow

### 2.5.1 Test Lifecycle

```
1. Test Collection Setup
   ??? Start game process (or attach to running game)
   ??? Establish automation channel
   ??? Wait for game ready

2. Test Fixture Setup
   ??? Navigate to required state
   ??? Create page objects

3. Individual Test
   ??? Arrange: Query current state
   ??? Act: Perform user actions
   ??? Assert: Verify expected state

4. Test Fixture Teardown
   ??? Reset to known state

5. Test Collection Teardown
   ??? Disconnect automation channel
   ??? Stop game process (if started)
```

### 2.5.2 Game Lifecycle Management

```csharp
public class StrideGameDriver : IDisposable
{
    private Process? _gameProcess;
    private IAutomationChannel? _channel;
    
    public async Task StartGameAsync(StrideTestOptions options)
    {
        // Build game executable path
        var gamePath = options.GameExecutablePath 
            ?? FindGameExecutable();
        
        // Start game with automation enabled
        _gameProcess = Process.Start(new ProcessStartInfo
        {
            FileName = gamePath,
            Arguments = "--enable-automation",
            UseShellExecute = false
        });
        
        // Wait for game window
        await WaitForGameWindowAsync(options.StartupTimeout);
        
        // Connect automation channel
        _channel = new NamedPipeChannel();
        await _channel.ConnectAsync(options.ConnectionTimeout);
    }
    
    public async Task StopGameAsync()
    {
        // Send exit command
        await _channel?.SendCommandAsync(new AutomationCommand 
        { 
            Type = "Action", 
            Method = "Exit" 
        });
        
        // Wait for clean exit
        _gameProcess?.WaitForExit(5000);
        
        // Force kill if needed
        if (!_gameProcess?.HasExited ?? false)
            _gameProcess?.Kill();
    }
}
```

```xml
<!-- Oravey.Game.csproj -->
<PropertyGroup Condition="'$(Configuration)' == 'Debug'">
  <DefineConstants>$(DefineConstants);ENABLE_AUTOMATION</DefineConstants>
</PropertyGroup>

<PropertyGroup Condition="'$(Configuration)' == 'UITest'">
  <DefineConstants>$(DefineConstants);ENABLE_AUTOMATION</DefineConstants>
</PropertyGroup>

<!-- Automation only in debug/test builds -->
<ItemGroup Condition="$(DefineConstants.Contains('ENABLE_AUTOMATION'))">
  <Compile Include="Automation\**\*.cs" />
</ItemGroup>
```

```
Debug      ? ENABLE_AUTOMATION (for local development + testing)
UITest     ? ENABLE_AUTOMATION (CI/CD test builds)
```
